cmake_minimum_required(VERSION 3.25)
project(SippingCloftware VERSION 0.1.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Build with native x64 MSVC on Windows")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
find_package(Qt6 6.10.0 EXACT REQUIRED COMPONENTS Core Gui Widgets Multimedia MultimediaWidgets Test Network)
set(FFMPEG_ROOT "${CMAKE_SOURCE_DIR}/.deps/ffmpeg/ffmpeg-master-latest-win64-gpl-shared" CACHE PATH "Pinned FFmpeg SDK")
add_library(clipper_core STATIC src/common.cpp src/registry.cpp src/media.cpp src/replay.cpp src/replay_storage.cpp src/capture.cpp src/audio.cpp src/recorder.cpp src/jobs.cpp src/platform.cpp src/library.cpp src/edit.cpp src/hdr.cpp src/compression.cpp src/notifications.cpp src/instance.cpp src/recycle.cpp)
target_include_directories(clipper_core PUBLIC src "${FFMPEG_ROOT}/include")
target_link_directories(clipper_core PUBLIC "${FFMPEG_ROOT}/lib")
target_compile_definitions(clipper_core PUBLIC UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
target_compile_options(clipper_core PUBLIC /W4 /EHsc /utf-8 /permissive- /Zc:__cplusplus)
target_link_libraries(clipper_core PUBLIC Qt6::Core Qt6::Gui Qt6::Network avcodec avformat avutil avfilter swscale swresample d3d11 dxgi dxguid ole32 shell32 propsys uuid runtimeobject dwmapi shlwapi mmdevapi)
set(CLIPPER_UI src/ui.cpp src/ui.h src/editor.cpp src/editor.h src/hud_backdrop.cpp src/hud_backdrop.h)
add_executable(clipper WIN32 src/main.cpp ${CLIPPER_UI})
set_target_properties(clipper PROPERTIES OUTPUT_NAME SippingCloftware)
target_link_libraries(clipper PRIVATE clipper_core Qt6::Widgets Qt6::Multimedia Qt6::MultimediaWidgets CoreMessaging)
add_executable(clipper_tests tests/core_tests.cpp)
target_link_libraries(clipper_tests PRIVATE clipper_core Qt6::Test)
add_executable(clipper_media_tests tests/media_tests.cpp)
target_link_libraries(clipper_media_tests PRIVATE clipper_core Qt6::Test)
add_executable(capture_fixture WIN32 tests/capture_fixture.cpp)
add_executable(clipboard_fixture tests/clipboard_fixture.cpp)
target_link_libraries(clipboard_fixture PRIVATE Qt6::Core user32)
add_executable(clipper_compression_tests tests/compression_tests.cpp)
target_link_libraries(clipper_compression_tests PRIVATE clipper_core Qt6::Test)
add_executable(clipper_notification_tests tests/notification_tests.cpp)
target_link_libraries(clipper_notification_tests PRIVATE clipper_core Qt6::Test)
target_link_libraries(capture_fixture PRIVATE Qt6::Widgets winmm)
add_executable(clipper_ui_tests tests/ui_tests.cpp ${CLIPPER_UI})
target_link_libraries(clipper_ui_tests PRIVATE clipper_core Qt6::Test Qt6::Widgets Qt6::Multimedia Qt6::MultimediaWidgets CoreMessaging)
enable_testing()
file(MAKE_DIRECTORY "${CMAKE_SOURCE_DIR}/test-output")
add_test(NAME core COMMAND clipper_tests -o "${CMAKE_SOURCE_DIR}/test-output/core.txt,txt")
add_test(NAME media COMMAND clipper_media_tests -o "${CMAKE_SOURCE_DIR}/test-output/media.txt,txt")
add_test(NAME compression COMMAND clipper_compression_tests -o "${CMAKE_SOURCE_DIR}/test-output/compression.txt,txt")
add_test(NAME notifications COMMAND clipper_notification_tests -o "${CMAKE_SOURCE_DIR}/test-output/notifications.txt,txt")
set_tests_properties(notifications PROPERTIES TIMEOUT 60)
set_tests_properties(compression PROPERTIES TIMEOUT 180)
set_tests_properties(media PROPERTIES TIMEOUT 120)
set_tests_properties(core PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 120)
foreach(scale 1 1.5 2)
  add_test(NAME ui_scale_${scale} COMMAND clipper_ui_tests -o "${CMAKE_SOURCE_DIR}/test-output/ui-${scale}.txt,txt")
  set_tests_properties(ui_scale_${scale} PROPERTIES ENVIRONMENT "QT_SCALE_FACTOR=${scale}" TIMEOUT 120 RUN_SERIAL TRUE WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}")
endforeach()
file(GLOB FFMPEG_DLLS "${FFMPEG_ROOT}/bin/*.dll")
foreach(target clipper clipper_tests)
  add_custom_command(TARGET ${target} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different ${FFMPEG_DLLS} "$<TARGET_FILE_DIR:${target}>")
endforeach()
add_custom_command(TARGET clipper_tests POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "$<TARGET_FILE:Qt6::Test>" "$<TARGET_FILE_DIR:clipper_tests>")
add_custom_command(TARGET clipper POST_BUILD
 COMMAND ${CMAKE_COMMAND} -E copy_if_different "${FFMPEG_ROOT}/bin/ffmpeg.exe" "${FFMPEG_ROOT}/bin/ffprobe.exe" "$<TARGET_FILE_DIR:clipper>"
 COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_SOURCE_DIR}/assets" "$<TARGET_FILE_DIR:clipper>/assets")
install(TARGETS clipper RUNTIME DESTINATION .)
